    char = input('Enter the letter: ')
    vowels = ['a', 'e', 'i', 'o', 'u', 'A', 'E', 'I', 'O', 'U']
    if char in vowels:
        print(char, ' is a vowel')
    else :
        print(char, ' is a consonant')
